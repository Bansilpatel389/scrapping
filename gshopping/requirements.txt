- name: Install Python dependencies
  run: |
    python -m pip install --upgrade pip setuptools wheel

    # Remove potentially cached/incompatible binaries
    pip uninstall -y numpy pandas || true

    # Install pinned compatible versions
    pip install --no-cache-dir -r gshopping/requirements.txt

    echo "Installed versions:"
    python -c "import numpy; print('NumPy:', numpy.__version__)"
    python -c "import pandas; print('Pandas:', pandas.__version__)"
    python -c "import selenium; print('Selenium:', selenium.__version__)"
